def get_display_kotatsus(self):
        out = []
        for k in self.get_kotatsus():
            loc = _call_or_attr(k, "location", (0,0,0))
            sz  = _call_or_attr(k, "size", (1,1))
            x, y = loc[0], loc[1]
            h, w = sz

            color = k.status.lower()
            if color == "empty":
                color = "green"
            elif color == "fill":
                color = "blue"
            else:
                color = "gray"

            out.append({
                "id": k.id,
                "shelfCode": k.id,
                "location_x": _as_float(x),
                "location_y": _as_float(y),
                "height": _as_float(h),
                "width": _as_float(w),
                "angle": k.angle,
                "color": color,
            })
        return out
